// ============================================================
// MiniC Grammar (LL(1) / Recursive Descent Ready)
// ============================================================

// --- Program Structure ---

program           ::= extern_list decl_list
                    | decl_list

extern_list       ::= extern extern_list
                    | epsilon

extern            ::= "extern" type_spec IDENT "(" params ")" ";"

// Decl List: Recursive definition
decl_list         ::= decl decl_list
                    | epsilon

// --- Declarations (Left Factored) ---
// ORIGINAL PROBLEM: var_decl ("int x;") and fun_decl ("int x(...)") 
// both started with "int x". The parser couldn't decide which path to take.
// FIX: Parse Type -> Parse Ident -> Then decide based on the next token.

decl              ::= type_spec IDENT decl_prime

decl_prime        ::= var_decl_suffix
                    | fun_decl_suffix

// Variable Declaration Suffix: Handles scalars (";") and arrays ("[5][5];")
var_decl_suffix   ::= ";"
                    | "[" INT_LIT "]" var_decl_suffix

// Function Declaration Suffix: Handles parameters and block
fun_decl_suffix   ::= "(" params ")" block

type_spec         ::= "void"
                    | var_type

var_type          ::= "int" | "float" | "bool"

// --- Function Parameters ---

params            ::= param_list
                    | "void" 
                    | epsilon

param_list        ::= param param_list_prime

param_list_prime  ::= "," param param_list_prime
                    | epsilon

// Array dimensions in params (e.g., int arr[][5])
param             ::= var_type IDENT param_array_dims

param_array_dims  ::= "[" "]" param_array_dims_rest  // First dim empty in params
                    | epsilon

param_array_dims_rest ::= "[" INT_LIT "]" param_array_dims_rest
                        | epsilon

// --- Block & Locals ---

block             ::= "{" local_decls stmt_list "}"

local_decls       ::= local_decl local_decls
                    | epsilon

// Re-uses the left-factored logic (parse type, then ident, then dimensions/semicolon)
local_decl        ::= var_type IDENT var_decl_suffix

// --- Statements ---

stmt_list         ::= stmt stmt_list
                    | epsilon

stmt              ::= expr_stmt
                    | block
                    | if_stmt
                    | while_stmt
                    | return_stmt

expr_stmt         ::= expr ";"
                    | ";"

while_stmt        ::= "while" "(" expr ")" stmt

if_stmt           ::= "if" "(" expr ")" block else_stmt

else_stmt         ::= "else" block
                    | epsilon

return_stmt       ::= "return" return_stmt_prime

return_stmt_prime ::= ";"
                    | expr ";"

// ============================================================
// Expressions (Precedence & Associativity)
// ============================================================
// Hierarchy (Lowest to Highest):
// 1. Assignment (=)
// 2. Logical OR (||)
// 3. Logical AND (&&)
// 4. Equality (==, !=)
// 5. Relational (<, <=, >, >=)
// 6. Additive (+, -)
// 7. Multiplicative (*, /, %)
// 8. Unary (-, !)
// 9. Primary (Identifiers, Literals, Calls, Array Access)

// --- 1. Assignment ---
// ORIGINAL PROBLEM: "IDENT = expr" vs "IDENT || ...".
// FIX: Parse the LHS (logical_or_expr). Then check if an "=" follows.
expr              ::= logical_or_expr expr_prime

expr_prime        ::= "=" expr
                    | epsilon

// --- 2. Logical OR ---
logical_or_expr   ::= logical_and_expr logical_or_prime

logical_or_prime  ::= "||" logical_and_expr logical_or_prime
                    | epsilon

// --- 3. Logical AND ---
logical_and_expr  ::= equality_expr logical_and_prime

logical_and_prime ::= "&&" equality_expr logical_and_prime
                    | epsilon

// --- 4. Equality ---
equality_expr     ::= relational_expr equality_prime

equality_prime    ::= "==" relational_expr equality_prime
                    | "!=" relational_expr equality_prime
                    | epsilon

// --- 5. Relational ---
relational_expr   ::= additive_expr relational_prime

relational_prime  ::= "<"  additive_expr relational_prime
                    | "<=" additive_expr relational_prime
                    | ">"  additive_expr relational_prime
                    | ">=" additive_expr relational_prime
                    | epsilon

// --- 6. Additive ---
additive_expr     ::= multiplicative_expr additive_prime

additive_prime    ::= "+" multiplicative_expr additive_prime
                    | "-" multiplicative_expr additive_prime
                    | epsilon

// --- 7. Multiplicative ---
multiplicative_expr ::= unary_expr multiplicative_prime

multiplicative_prime ::= "*" unary_expr multiplicative_prime
                       | "/" unary_expr multiplicative_prime
                       | "%" unary_expr multiplicative_prime
                       | epsilon

// --- 8. Unary ---
unary_expr        ::= "!" unary_expr
                    | "-" unary_expr
                    | primary_expr

// --- 9. Primary ---
// Handles Identifiers, Function Calls, and Arrays via Left Factoring.
primary_expr      ::= IDENT ident_suffix
                    | INT_LIT
                    | FLOAT_LIT
                    | BOOL_LIT
                    | "(" expr ")"

// Decides if IDENT is a variable, a function call, or an array access
ident_suffix      ::= "(" args ")"        // Function Call
                    | array_access_suffix // Array Access or Scalar Variable
                    | epsilon             // Scalar Variable

array_access_suffix ::= "[" expr "]" array_access_suffix
                      | epsilon

// --- Arguments ---
args              ::= arg_list
                    | epsilon

arg_list          ::= expr arg_list_prime

arg_list_prime    ::= "," expr arg_list_prime
                    | epsilon
